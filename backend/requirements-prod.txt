# Dependency versions for a hosted deploy.
#
# Tests are deliberately excluded here; they live in requirements-dev.txt so the
# deployed image stays small. pip does not understand docstrings, so comments
# must use '#'.

fastapi==0.115.6
uvicorn[standard]==0.34.0
sqlalchemy==2.0.36
pydantic==2.10.4
pydantic-settings==2.7.0
python-multipart==0.0.20
PyJWT==2.10.1
email-validator==2.2.0
httpx==0.28.1
redis==8.1.0

# Postgres driver for a hosted database. SQLite needs nothing extra, so this is
# unused in local development but required when DATABASE_URL points at Postgres.
psycopg[binary]==3.2.3

# Web Push: signs and delivers push messages to browser push services (FCM,
# Mozilla autopush, WNS). Imported lazily by app.push, so a deployment that does
# not set VAPID keys still boots without it.
pywebpush==2.0.3

# Composes the Open Graph preview card served to link-preview crawlers
# (/api/preview/card/{username}). Imported lazily by app.routers.preview, so the
# app still boots without it and only the card endpoint degrades.
pillow==13.0.0

# Language detection for post translation. A local, pure-Python classifier with
# no network cost, imported lazily by app.translate, so the app boots without it
# and simply offers no translate affordance.
langdetect==1.0.9

# Video preview frames. There is no pure-Python H.264 decoder, so the server
# needs ffmpeg to produce a poster for a video whose browser capture did not
# happen. This wheel bundles the binary (the host has none), which is what makes
# the server-side fallback real rather than a hoped-for `ffmpeg` on PATH.
# Imported lazily by app.video, so a deployment without it still boots and
# simply falls back to the client-side capture.
imageio-ffmpeg==0.8.1
